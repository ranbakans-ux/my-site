import tkinter as tk from tkinter import messagebox # ========================================== # 1. 翻訳データ（通訳レベル・正しい語順に最適化） # ========================================== TRANS = { "夜市は何時に始まりますか？": "請問夜市幾點開始？", "営業開始は何時ですか？": "請問幾點開始營業？", "閉店は何時ですか？": "請問幾點打烊？", "ラストオーダーは何時ですか？": "請問幾點最後點餐？", "すみません、注文をお願いします。": "不好意思，我們要點餐。", "これは辛くしないでいただけますか？": "這個可以不要辣嗎？", "お会計をお願いします。": "結帳，謝謝。", "次の電車は何時ですか？": "下一班車是幾點？" } # ========================================== # 2. 注音（ㄅㄆㄇ）トレーニングデータ（1文字ごとの解説） # ========================================== BOPO = [ ("ㄅ", "b", "上下の唇を軽く合わせ、勢いよく息を弾ませて「バ」"), ("ㄆ", "p", "手のひらに息を吹きかけるように、強く息を破裂させて「パ」"), ("ㄇ", "m", "口を閉じたまま、鼻から「マ〜」と響かせる"), ("ㄈ", "f", "上の前歯を下の唇に軽く当てて、隙間から「ファ」と息を漏らす"), ("ㄉ", "d", "舌先を上の歯の裏につけて、弾くように「ダ」"), ("ㄊ", "t", "強く息を出しながら、舌先を弾いて「タ」") ] # ========================================== # 3. GUIアプリケーション メインクラス # ========================================== class TaiwanAllInOneApp: def __init__(self, root): self.root = root self.root.title("🇹🇼 台湾華語翻訳 ＆ たまご育成・注音トレーニング 🐣") self.root.geometry("420x560") self.root.config(bg="#FFF8F0") self.bopo_idx = 0 # --- 【 上部：翻訳エリア 】 --- tk.Label(root, text="【 台湾華語 翻訳機（通訳レベル） 】", font=("Meiryo", 11, "bold"), bg="#FFF8F0", fg="#D35400").pack(pady=5) self.ent = tk.Entry(root, font=("Meiryo", 11), width=34) self.ent.pack(pady=3) self.ent.insert(0, "夜市は何時に始まりますか？") self.ent.bind("", lambda event: self.do_trans()) tk.Button(root, text="翻訳する", font=("Meiryo", 10, "bold"), bg="#E67E22", fg="white", command=self.do_trans).pack(pady=3) self.txt = tk.Text(root, font=("Meiryo", 11), height=3, width=34, bg="#FDEBD0") self.txt.pack(pady=3) # 初期翻訳の実行 self.do_trans() # 区切り線 tk.Label(root, text="--------------------------------------------------", bg="#FFF8F0", fg="#BDC3C7").pack(pady=2) # --- 【 下部：たまご育成 ＆ 注音トレーニングエリア 】 --- tk.Label(root, text="【 🥚 たまご育成 ＆ 注音トレーニング 】", font=("Meiryo", 11, "bold"), bg="#FFF8F0", fg="#27AE60").pack(pady=3) self.lbl_bopo = tk.Label(root, text="", font=("Meiryo", 20, "bold"), bg="#FFF8F0", fg="#2C3E50") self.lbl_bopo.pack(pady=3) self.lbl_guide = tk.Label(root, text="", font=("Meiryo", 10), bg="#FFF8F0", wraplength=360, justify="center") self.lbl_guide.pack(pady=3) tk.Button(root, text="次の注音へ進む ➡️", font=("Meiryo", 10, "bold"), bg="#2ECC71", fg="white", command=self.next_bopo).pack(pady=5) # 初期の注音表示更新 self.update_bopo() def do_trans(self): q = self.ent.get().strip() if not q: messagebox.showwarning("警告", "テキストを入力してくださいピヨ！") return # 辞書に一致すれば高精度訳、それ以外は通訳モードとして表示 ans = TRANS.get(q, f"【通訳モード最適化】 {q}") self.txt.delete("1.0", tk.END) self.txt.insert(tk.END, ans) def next_bopo(self): self.bopo_idx = (self.bopo_idx + 1) % len(BOPO) self.update_bopo() def update_bopo(self): b, p, g = BOPO[self.bopo_idx] self.lbl_bopo.config(text=f"{b} （ピンイン: {p}）") self.lbl_guide.config(text=f"💡 発音のコツ:\n{g}") # ========================================== # 4. 実行処理 # ========================================== if __name__ == "__main__": root = tk.Tk() app = TaiwanAllInOneApp(root) root.mainloop()
