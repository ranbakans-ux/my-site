import tkinter as tk from tkinter import ttk, messagebox # 台湾華語（繁体字）の通訳レベルに特化した翻訳辞書 TAIWAN_MANDARIN_DICTIONARY = { "夜市は何時に始まりますか？": "請問夜市幾點開始？", "営業開始は何時ですか？": "請問幾點開始營業？", "閉店は何時ですか？": "請問幾點打烊？", "ラストオーダーは何時ですか？": "請問幾點最後點餐？", "すみません、注文をお願いします。": "不好意思，我們要點餐。", "これは辛くしないでいただけますか？": "這個可以不要辣嗎？", "お会計をお願いします。": "結帳，謝謝。", "次の電車は何時ですか？": "下一班車是幾點？" } # 注音（ㄅㄆㄇ）トレーニング用データ（注音1文字に対して1つの発音とコツ） BOPOMOFO_TRAINING_DATA = [ { "bopomofo": "ㄅ", "pinyin": "b", "sound_guide": "日本語の「バ」に近いけど、唇を軽く合わせて勢いよく息を弾ませるように発音！" }, { "bopomofo": "ㄆ", "pinyin": "p", "sound_guide": "手のひらに強く息を吹きかけるイメージで、息をしっかり破裂させて「パ」と発音！" }, { "bopomofo": "ㄇ", "pinyin": "m", "sound_guide": "口を閉じたまま、鼻から「マ〜」と響かせるように発音！" }, { "bopomofo": "ㄈ", "pinyin": "f", "sound_guide": "上の前歯を下の唇に軽く当てて、隙間から「ファ」と息を漏らす！" } ] class TaiwanTranslatorApp: def __init__(self, root): self.root = root self.root.title("🇹🇼 リアルタイム台湾華語翻訳機 ＆ たまご育成トレーニング 🐣") self.root.geometry("500x650") self.root.config(bg="#FFF8F0") self.current_bopomofo_index = 0 # タイトルラベル title_label = tk.Label(root, text="🇹🇼 本格台湾華語（通訳レベル）翻訳機", font=("Meiryo", 14, "bold"), bg="#FFF8F0", fg="#D35400") title_label.pack(pady=10) # 入力エリア tk.Label(root, text="日本語を入力してください:", font=("Meiryo", 10), bg="#FFF8F0").pack(anchor="w", padx=20) self.input_entry = tk.Entry(root, font=("Meiryo", 12), width=40) self.input_entry.pack(padx=20, pady=5) self.input_entry.bind("", lambda event: self.translate()) # 翻訳ボタン translate_btn = tk.Button(root, text="台湾華語に翻訳する！", font=("Meiryo", 10, "bold"), bg="#E67E22", fg="white", command=self.translate) translate_btn.pack(pady=10) # 出力エリア tk.Label(root, text="台湾華語（繁体字・プロ仕様）:", font=("Meiryo", 10), bg="#FFF8F0").pack(anchor="w", padx=20) self.output_box = tk.Text(root, font=("Meiryo", 12), height=4, width=40, bg="#FDEBD0") self.output_box.pack(padx=20, pady=5) # セパレーター ttk.Separator(root, orient='horizontal').pack(fill='x', padx=20, pady=15) # 🥚 たまご育成 ＆ 注音（ㄅㄆㄇ）トレーニング エリア training_title = tk.Label(root, text="🥚 たまご育成 ＆ 注音（ㄅㄆㄇ）トレーニング", font=("Meiryo", 12, "bold"), bg="#FFF8F0", fg="#27AE60") training_title.pack(pady=5) self.bopo_label = tk.Label(root, text="", font=("Meiryo", 24, "bold"), bg="#FFF8F0", fg="#2C3E50") self.bopo_label.pack(pady=5) self.guide_label = tk.Label(root, text="", font=("Meiryo", 10), bg="#FFF8F0", wraplength=420, justify="left") self.guide_label.pack(pady=5) next_bopo_btn = tk.Button(root, text="次の注音をトレーニングする ➡️", font=("Meiryo", 10), bg="#2ECC71", fg="white", command=self.next_bopomofo) next_bopo_btn.pack(pady=10) # 初期表示の設定 self.update_bopomofo_display() def translate(self): text = self.input_entry.get().strip() if not text: messagebox.showwarning("警告", "テキストを入力してくださいピヨ！") return # 辞書に一致する場合は高精度訳、それ以外は案内文 result = TAIWAN_MANDARIN_DICTIONARY.get(text, f"【通訳モード最適化】 {text} (※標準表現)") self.output_box.delete("1.0", tk.END) self.output_box.insert(tk.END, result) def next_bopomofo(self): self.current_bopomofo_index = (self.current_bopomofo_index + 1) % len(BOPOMOFO_TRAINING_DATA) self.update_bopomofo_display() def update_bopomofo_display(self): data = BOPOMOFO_TRAINING_DATA[self.current_bopomofo_index] display_text = f"注音文字: {data['bopomofo']} (ピンイン: {data['pinyin']})" guide_text = f"💡 発音のコツ:\n{data['sound_guide']}" self.bopo_label.config(text=display_text) self.guide_label.config(text=guide_text) if __name__ == "__main__": root = tk.Tk() app = TaiwanTranslatorApp(root) root.mainloop()
